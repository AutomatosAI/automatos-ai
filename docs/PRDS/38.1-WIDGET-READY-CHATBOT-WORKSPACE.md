## Executive Summary

Transform the current chatbot interface from a modal-based artifact viewer into a **canvas-based workspace** where widgets (artifacts) persist, can be arranged, and remain accessible throughout the conversation. This phase establishes the widget architecture foundation that enables all future phases.

```

        │
        ▼

        │
        ▼

        │
        ▼
useChat hook receives event
        │
        ▼

        │
        ▼

        │
        ▼

        │
```

```typescript
// frontend/stores/workspace-store.ts
import { create } from 'zustand'
import { persist } from 'zustand/middleware'

interface WorkspaceState {
  // Widgets
  widgets: Record<string, Widget>
  widgetIds: string[]  // Ordered list for rendering
  activeWidgetId: string | null

  // Layout
  layoutMode: 'grid' | 'freeform' | 'split' | 'focus'
  gridConfig: GridConfig
  positions: Record<string, WidgetPosition>
  sizes: Record<string, WidgetSize>

  // Chat integration
  chatPanelWidth: number
  isChatCollapsed: boolean
  widgetsInContext: string[]

  // UI state
  isWidgetTrayOpen: boolean
  selectedWidgetForContext: string | null
}

interface WorkspaceActions {
  // Widget CRUD
  addWidget: (widget: Omit<Widget, 'id'>) => string
  updateWidget: (id: string, updates: Partial<Widget>) => void
  removeWidget: (id: string) => void
  clearWidgets: () => void

  // Selection
  setActiveWidget: (id: string | null) => void
  bringToFront: (id: string) => void
  sendToBack: (id: string) => void

  // Layout
  setLayoutMode: (mode: LayoutMode) => void
  updatePosition: (id: string, position: WidgetPosition) => void
  updateSize: (id: string, size: WidgetSize) => void
  resetLayout: () => void

  // Chat panel
  setChatPanelWidth: (width: number) => void
  toggleChatCollapsed: () => void

  // Context (for chat references)
  addToContext: (widgetId: string) => void
  removeFromContext: (widgetId: string) => void
  clearContext: () => void

  // Tray
  toggleWidgetTray: () => void

  // From tool results
  handleToolResult: (
    toolName: string,
    toolCallId: string,
    result: any,
    conversationId: string
  ) => void
}

export const useWorkspaceStore = create<WorkspaceState & WorkspaceActions>()(
  persist(
    (set, get) => ({
      // Initial state
      widgets: {},
      widgetIds: [],
      activeWidgetId: null,
      layoutMode: 'grid',
      gridConfig: { columns: 12, rowHeight: 100 },
      positions: {},
      sizes: {},
      chatPanelWidth: 400,
      isChatCollapsed: false,
      widgetsInContext: [],
      isWidgetTrayOpen: true,
      selectedWidgetForContext: null,

      // Actions implementation...
      addWidget: (widgetData) => {
        // Use crypto.randomUUID() for collision-resistant IDs
        // Falls back to nanoid pattern if crypto.randomUUID unavailable
        const id = typeof crypto !== 'undefined' && crypto.randomUUID
          ? `widget-${crypto.randomUUID()}`
          : `widget-${Date.now().toString(36)}-${Array.from(
              crypto.getRandomValues(new Uint8Array(8)),
              b => b.toString(16).padStart(2, '0')
            ).join('')}`

        const widget: Widget = {
          ...widgetData,
          id,
          state: 'ready',
          createdAt: new Date().toISOString(),
        }

        set((state) => ({
          widgets: { ...state.widgets, [id]: widget },
          widgetIds: [...state.widgetIds, id],
          activeWidgetId: id,
        }))

        return id
      },

      handleToolResult: (toolName, toolCallId, result, conversationId) => {
        const widgetType = routeToolToWidget(toolName, result)
        const widgetData = transformToolResultToWidget(
          widgetType,
          toolName,
          result,
          conversationId,
          toolCallId
        )

        if (widgetData) {
          get().addWidget(widgetData)
        }
      },

      // ... other actions
    }),
    {
      name: 'automatos-workspace',
      partialize: (state) => ({
        layoutMode: state.layoutMode,
        chatPanelWidth: state.chatPanelWidth,
        isChatCollapsed: state.isChatCollapsed,
        isWidgetTrayOpen: state.isWidgetTrayOpen,
      }),
    }
  )
)
```

```typescript
// frontend/components/widgets/router.ts

import { WidgetType } from './types'

const TOOL_WIDGET_MAP: Record<string, WidgetType> = {
  // RAG Tools
  'search_knowledge': 'document',
  'semantic_search': 'document',
  'search_documents': 'document',
  'search_multimodal': 'document',

  // Database Tools
  'smart_query_database': 'data',
  'query_database': 'data',
  'execute_sql': 'data',

  // Code Tools
  'search_codebase': 'code',
  'get_code_context': 'code',
  'analyze_code': 'code',

  // Image Tools
  'generate_image': 'image',
  'edit_image': 'image',
  'analyze_image': 'image',
  'search_images': 'image',
}

export function routeToolToWidget(
  toolName: string,
  result: any
): WidgetType | null {
  // Direct mapping
  if (TOOL_WIDGET_MAP[toolName]) {
    return TOOL_WIDGET_MAP[toolName]
  }

  // Infer from result structure
  if (result.database_results?.length > 0) return 'data'
  if (result.documents?.length > 0) return 'document'
  if (result.code_snippets?.length > 0) return 'code'
  if (result.images?.length > 0) return 'image'

  // Check for specific data patterns
  if (result.sql || result.columns) return 'data'
  if (result.code || result.language) return 'code'
  if (result.content && (result.chunks || result.similarity)) return 'document'

  return null
}

export function transformToolResultToWidget(
  type: WidgetType,
  toolName: string,
  result: any,
  conversationId: string,
  toolCallId: string
): Omit<Widget, 'id'> | null {
  const baseWidget = {
    type,
    metadata: {
      source: {
        type: 'tool' as const,
        name: toolName,
      },
      conversationId,
      toolCallId,
    },
    state: 'ready' as const,
  }

  switch (type) {
    case 'data':
      return {
        ...baseWidget,
        title: result.database || 'Query Result',
        data: {
          columns: result.columns || [],
          rows: result.data || result.rows || [],
          sql: result.sql,
          database: result.database,
          rowCount: result.row_count || result.data?.length || 0,
          executionTime: result.execution_time_ms,
          charts: result.pandas_ai?.charts,
          pandasAiSummary: result.pandas_ai?.summary,
        },
      }

    case 'document':
      return {
        ...baseWidget,
        title: result.title || result.filename || 'Document',
        data: {
          content: result.full_content || result.content || result.preview,
          format: 'markdown',
          filename: result.filename,
          filePath: result.file_path,
          similarity: result.relevance || result.similarity,
          chunkCount: result.chunk_count,
          chunks: result.chunks,
          downloadUrl: result.download_url,
        },
      }

    case 'code':
      return {
        ...baseWidget,
        title: result.symbol_name || result.file_path || 'Code',
        data: {
          code: result.code || result.content,
          language: result.language || 'python',
          filePath: result.file_path,
          lineNumber: result.line_number,
          explanation: result.explanation,
          symbolName: result.symbol_name,
        },
      }

    case 'image':
      return {
        ...baseWidget,
        title: result.filename || 'Image',
        data: {
          src: result.base64
            ? `data:${result.mime_type || 'image/png'};base64,${result.base64}`
            : result.url || result.src,
          alt: result.alt || result.prompt,
          mimeType: result.mime_type,
          prompt: result.prompt,
          model: result.model,
        },
      }

    default:
      return null
  }
}
```

```typescript
// frontend/components/widgets/WidgetWrapper.tsx

import { useCallback } from 'react'
import { Widget, getWidget } from './registry'
import { WidgetBase } from './WidgetBase'
import { useWorkspaceStore } from '@/stores/workspace-store'

interface WidgetWrapperProps {
  widget: Widget
  isActive?: boolean
}

export function WidgetWrapper({ widget, isActive }: WidgetWrapperProps) {
  const removeWidget = useWorkspaceStore((s) => s.removeWidget)
  const setActiveWidget = useWorkspaceStore((s) => s.setActiveWidget)

  const definition = getWidget(widget.type)

  const handleClose = useCallback(() => {
    removeWidget(widget.id)
  }, [widget.id, removeWidget])

  const handleClick = useCallback(() => {
    setActiveWidget(widget.id)
  }, [widget.id, setActiveWidget])

  if (!definition) {
    return (
      <WidgetBase
        title="Unknown Widget"
        isActive={isActive}
        error={new Error(`Unknown widget type: ${widget.type}`)}
        onClose={handleClose}
      >
        <div />
      </WidgetBase>
    )
  }

  const WidgetComponent = definition.component

  return (
    <div onClick={handleClick} className="h-full">
      <WidgetComponent
        id={widget.id}
        title={widget.title}
        data={widget.data}
        metadata={widget.metadata}
        isActive={isActive}
        isLoading={widget.state === 'loading'}
        error={widget.state === 'error' ? new Error('Widget error') : null}
        onClose={handleClose}
      />
    </div>
  )
}
```

```typescript
  const widgetIds = useWorkspaceStore((s) => s.widgetIds)
  const activeWidgetId = useWorkspaceStore((s) => s.activeWidgetId)
  const positions = useWorkspaceStore((s) => s.positions)
  const sizes = useWorkspaceStore((s) => s.sizes)
  const updatePosition = useWorkspaceStore((s) => s.updatePosition)
  const updateSize = useWorkspaceStore((s) => s.updateSize)

  // Convert our state to react-grid-layout format
  const layout: Layout[] = useMemo(() => {
    return widgetIds.map((id, index) => {
      const pos = positions[id] || { x: (index % 2) * 6, y: Math.floor(index / 2) * 4 }
      const size = sizes[id] || { width: 6, height: 4 }

        i: id,
        x: pos.x,
        y: pos.y,
        w: size.width,
        h: size.height,
        minW: 3,
        minH: 2,
      }

      const currentSize = sizes[item.i]

      if (!currentPos || currentPos.x !== item.x || currentPos.y !== item.y) {
        updatePosition(item.i, { x: item.x, y: item.y })
      }

      if (!currentSize || currentSize.width !== item.w || currentSize.height !== item.h) {
        updateSize(item.i, { width: item.w, height: item.h })
      }
    })
  }, [positions, sizes, updatePosition, updateSize])

  if (widgetIds.length === 0) {
    return (
```

### 9.2 Widget Reference Syntax

**Decision:** Use `@widget-<short-id>` pattern with the following rules:

**Syntax Rules:**
```
@widget-<short-id>    # Reference by short ID (first 8 chars of UUID)
@widget:title         # Reference by title (fuzzy match)
@widget-latest        # Reference most recently created widget
@widget-active        # Reference currently active widget
```

**Parsing Implementation:**
```typescript
// lib/widgets/reference-parser.ts

const WIDGET_REF_PATTERN = /@widget-([a-f0-9]{8}|latest|active)|@widget:([^\s]+)/gi

interface WidgetReference {
  type: 'id' | 'title' | 'special'
  value: string
  originalText: string
}

function parseWidgetReferences(text: string): WidgetReference[] {
  const matches = [...text.matchAll(WIDGET_REF_PATTERN)]
  return matches.map(match => ({
    type: match[1] ? (match[1] === 'latest' || match[1] === 'active' ? 'special' : 'id') : 'title',
    value: match[1] || match[2],
    originalText: match[0]
  }))
}

function resolveWidgetReference(ref: WidgetReference, store: WorkspaceState): Widget | null {
  switch (ref.type) {
    case 'id':
      return Object.values(store.widgets).find(w => w.id.startsWith(`widget-${ref.value}`)) || null
    case 'title':
      return Object.values(store.widgets).find(w =>
        w.title.toLowerCase().includes(ref.value.toLowerCase())
      ) || null
    case 'special':
      if (ref.value === 'latest') {
        return store.widgetIds.length > 0
          ? store.widgets[store.widgetIds[store.widgetIds.length - 1]]
          : null
      }
      if (ref.value === 'active') {
        return store.activeWidgetId ? store.widgets[store.activeWidgetId] : null
      }
      return null
  }
}
```

**Scope:** References are resolved within the current workspace/conversation context.

### 9.3 Mobile Layout Strategy

**Decision:** Use responsive breakpoints with simplified layout on mobile.

**Strategy:**
- **Desktop (≥1024px):** Full React Grid Layout with drag-drop
- **Tablet (768-1023px):** Simplified 2-column grid, drag-drop disabled
- **Mobile (<768px):** Single-column stack with swipeable widget cards

**Implementation:**
```typescript
// components/workspace/Canvas.tsx

import { useMediaQuery } from '@/hooks/useMediaQuery'

function Canvas() {
  const isMobile = useMediaQuery('(max-width: 767px)')
  const isTablet = useMediaQuery('(min-width: 768px) and (max-width: 1023px)')

  if (isMobile) {
    return <MobileWidgetStack widgets={widgets} />  // Swipeable cards
  }

  if (isTablet) {
    return (
      <ResponsiveGridLayout
        cols={2}
        isDraggable={false}
        isResizable={false}
        // ... simplified layout
      />
    )
  }

  // Desktop: full functionality
  return (
    <ResponsiveGridLayout
      cols={12}
      isDraggable={true}
      isResizable={true}
      // ... full layout
    />
  )
}

// Mobile-specific component
function MobileWidgetStack({ widgets }: { widgets: Widget[] }) {
  const [activeIndex, setActiveIndex] = useState(0)

  return (
    <div className="mobile-widget-stack">
      <div className="widget-indicators">
        {widgets.map((_, i) => (
          <button
            key={i}
            className={cn('indicator', i === activeIndex && 'active')}
            onClick={() => setActiveIndex(i)}
          />
        ))}
      </div>
      <SwipeableViews index={activeIndex} onChangeIndex={setActiveIndex}>
        {widgets.map(widget => (
          <WidgetWrapper key={widget.id} widget={widget} isMobile />
        ))}
      </SwipeableViews>
    </div>
  )
}
```

**Rationale:** React Grid Layout is not mobile-friendly due to drag-drop interaction patterns. A swipeable card stack provides better UX on touch devices.

**Alternative Considered:** Using a modal/drawer pattern on mobile. Rejected because it breaks the multi-widget visibility goal.
